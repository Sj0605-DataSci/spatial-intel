class PVAR:
    def __init__(self, name, args, info):
        self.name = name
        self.args = args
        self.fluent_type = info[0]
        self.type = info[1]
        self.default = info[2]
    
    def __str__(self):
        if (len(self.args) > 0): 
            return f"{self.name}({','.join(self.args)}) : {{ {self.fluent_type} , {self.type} , {self.default} }}"
        else:
            return f"{self.name} : {{ {self.fluent_type} , {self.type} , {self.default} }}"


class Branch:
    def __init__(self):
        self.conds = [] 
        self.rhs = None
        self.exist_str = None
    
    def _set_rhs(self, rhs):
        self.rhs = rhs 
    
    def _set_exist_str(self, exist_str):
        self.exist_str = exist_str

    def _add_cond(self, conds): 
        '''
            conds is a list of cond
            cond is a list:
                if pvar then cond = [pvar_name, arg1, arg2, ...] where arg1 etc are indexes in args of pvar
                if exist_str then cond = [exist_str]
        '''
        self.conds.extend(conds) 
    
    def _process_cond(self, cpfs):
        ret = [] 
        # print("pvar is ", cpfs.pvar) 
        # print("args is ", cpfs.args) 
        try:
            for cond in self.conds:
                '''
                    this function is needed to convert the cond to a string
                    if cond is of form [pvar_name, arg1, arg2, ...] then convert it to pvar_name(?arg1, ?arg2, ...)
                    arg1 etc are indexes, mapped to strs using cpfs.args
                '''
                to_add = cond[0] 
                if (len(cond) > 1):
                    to_add += "( "
                    for i in range(1, len(cond)):
                        to_add += f"?{cpfs.args[cond[i]]}, "
                    to_add = to_add[:-2]
                    to_add += ") " 

                ret.append(to_add) 
        except:
            print("conds is ", self.conds) 
            exit(1) 
        self.conds = ret 
    
    def __str__(self):
        ret = f"if ("
        if (len(self.conds) > 0):
            if (self.exist_str != None):
                ret += f"{self.exist_str}[ "
            cond_str = " & ".join(self.conds) 
            ret += f"{cond_str} "
            if (self.exist_str != None):
                ret += "] "
            ret += ") "
            ret += f" then {self.rhs}"
            return ret
        else:
            return f"{self.rhs}"
        

class CPFS:
    def __init__(self, pvar):
        self.pvar = pvar 
        self.branches = [] 
        if ("(" in self.pvar):
            self.args = self.pvar.split("(")[1].split(")")[0].split(",") 
            self.args = [arg.strip() for arg in self.args]
            self.pvar = self.pvar.split("(")[0]
        else:
            self.args = []
    
    def _add_branch(self, branch):
        branch._process_cond(self) 
        self.branches.append(branch)  
    
    def __str__(self):
        pvar = self.pvar + "'"
        
        # replace args with ?args
        for i in range(len(self.args)):
            self.args[i] = f"?{self.args[i]}"
        
        if (len(self.args) > 0):
            pvar += "(" + ",".join(self.args) + ")"

        ret = f"{pvar} = "
        ret += str(self.branches[0]) + "\n"
        for branch in self.branches[1:]:
            ret += f" else {str(branch)}" + "\n"
        ret += ";\n" 
        return ret
    


class YAML_to_RDDL:
    '''
        converts a yaml file to a rddl file for use by planner

    def _make_state_variables_unique(self):
        import re
        defined_bases = set()
        
        # Track bases of state variables
        for sv in self.parsed_data.get("state_variables", []):
            if "(" in sv:
                defined_bases.add(sv.split("(")[0].strip())
            else:
                defined_bases.add(sv.strip())

        # 1. Update state_variables dictionary
        new_sv = {}
        for sv, val in self.parsed_data.get("state_variables", {}).items():
            if "(" in sv:
                base = sv.split("(")[0].strip()
                args = [a.strip() for a in sv.split("(")[1].split(")")[0].split(",")]
                # e.g., is_collected -> is_collected_scripture
                new_base = f"{base}_{'_'.join([a.lower() for a in args])}"
                new_sv[f"{new_base}({', '.join(args)})"] = val
            else:
                new_sv[sv] = val
        self.parsed_data["state_variables"] = new_sv

        def rename_in_string(s):
            if not isinstance(s, str): return s
            matches = re.finditer(r'([a-zA-Z0-9_]+)\(([a-zA-Z0-9_\?,\s]+)\)', s)
            out = s
            for m in matches:
                full = m.group(0)
                base = m.group(1)
                
                # Only modify if it is a known overloaded state variable
                if base not in defined_bases:
                    continue
                    
                args_str = m.group(2)
                args = [a.strip() for a in args_str.split(",")]
                
                arg_classes = []
                for a in args:
                    clean_a = a.replace('?', '').lower()
                    found_class = False
                    for c_name, c_info in self.parsed_data.get("classes", {}).items():
                        if clean_a in [ins.lower() for ins in c_info.get("instances", [])]:
                            arg_classes.append(c_name.lower())
                            found_class = True
                            break
                    if not found_class:
                        # Fallback: assume the variable name matches the class name
                        arg_classes.append(clean_a)
                
                new_base = f"{base}_{'_'.join(arg_classes)}"
                out = out.replace(full, f"{new_base}({args_str})")
            return out

        # 2. Update initial_state_variables
        new_init = {}
        for sv, val in self.parsed_data.get("initial_state_variables", {}).items():
            new_init[rename_in_string(sv)] = val
        self.parsed_data["initial_state_variables"] = new_init

        # 3. Update transitions
        for t in self.parsed_data.get("transitions", []):
            if "preconditions" in t:
                t["preconditions"] = [rename_in_string(p) for p in t["preconditions"]]
            if "effects" in t:
                t["effects"] = [rename_in_string(e) for e in t["effects"]]
                
        # 4. Update goal and reward
        if "goal" in self.parsed_data:
            self.parsed_data["goal"] = rename_in_string(self.parsed_data["goal"])
        if "reward" in self.parsed_data:
            self.parsed_data["reward"] = rename_in_string(self.parsed_data["reward"])

                branch._add_cond([[f"move-down-{clss}", 0], [f"~exists_{{?y2:ypos}}[{clss}-y(?{clss}, ?y2) & ADJACENT-DOWN(?y2, ?ypos)]"] ])
                branch._set_rhs("false")
                cpfs._add_branch(branch)

                branch = Branch()
                branch._set_rhs(f"{clss}-y(?{clss}, ?ypos)")
                cpfs._add_branch(branch)
                cpfss.append(cpfs)

                # has moved action fluent
                cpfs = CPFS(f"has-moved-{clss}({clss})") 
                branch = Branch()
                branch._set_rhs(f"move-left-{clss}(?{clss}) | move-right-{clss}(?{clss}) | move-up-{clss}(?{clss}) | move-down-{clss}(?{clss})") 
                cpfs._add_branch(branch)
                cpfss.append(cpfs)

       

        
        sv_cpfs = dict()
        for sv, val in self.parsed_data["state_variables"].items():
            cpfs = CPFS(sv)
            sv_cpfs[sv] = cpfs

        # Build base-name lookup for state variables with arguments.
        # sv_cpfs keys use type names: "is_collected_clue(Clue)"
        # but transition effects use ?variables: "is_collected_clue(?clue)"
        # This mapping lets us match by base name when the full key doesn't match.
        sv_base_to_key = {}
        for key in sv_cpfs:
            base = key.split("(")[0] if "(" in key else key
            sv_base_to_key[base] = key

        # Build action signature lookup from the "actions" list so we can
        # resolve transition argument names (which may be abstract like
        # "participant") back to their canonical class-typed names (like
        # "festival_participants").
        action_signatures = {}
        for action in self.parsed_data.get("actions", []):
            if "(" not in action: continue
            a_name = action.split("(")[0].strip()
            a_args = action.split("(")[1].split(")")[0].split(",")
            a_args = [arg.strip().replace('?', '') for arg in a_args if arg.strip()]
            action_signatures[a_name.lower()] = a_args

        # prepare the condition. then for each effect, add the condition and effect to corresponding cpfs
        for transition in self.parsed_data.get("transitions", []):
            precond = []
            if ("(" not in transition["action"]):
                continue
            action_args = transition["action"].split("(")[1].split(")")[0].split(",")
            action_args = [action_arg.strip().replace('?', '') for action_arg in action_args]
            action_name = transition["action"].split("(")[0].strip()

            if action_name.lower().startswith("move"):
                continue

            # Resolve transition args to canonical class names from actions list
            canonical_args = action_signatures.get(action_name.lower(), action_args)
            # Build mapping: transition_arg -> canonical_class_type
            arg_to_class = {}
            for i, t_arg in enumerate(action_args):
                if i < len(canonical_args):
                    arg_to_class[t_arg.lower()] = canonical_args[i].lower()

            # Use canonical args for the PVAR name (must match the action fluent definition)
            pvar_base_name = "-".join([action_name] + [a.lower() for a in canonical_args])
            pre = f"{pvar_base_name}" # Action fluent, no prime (') required

            to_add_action_args = ["?"+arg.lower() for arg in action_args]
            arg_str = "(" + ", ".join(to_add_action_args) + ")"
            action_cond = pre + arg_str
            precond.append([action_cond])

            probability = transition.get("probability", None)
            is_redundant = False

            for prec in transition["preconditions"]:
                if (type(prec) == bool):
                    if (prec == True): continue
                    elif (prec == False):
                        is_redundant = True
                        break
                if (prec == None): continue
                if (prec.lower() == "true"): continue
                elif (prec.lower() == "false"):
                    is_redundant = True
                    break
                
                prec = self.modify_prec(prec, arg_to_class)
                precond.append([prec])
            
            if (is_redundant): continue

            for effect in transition["effects"]:
                if (effect is None): continue
                if ("=" not in effect): continue
                state_var = effect.split("=")[0].strip() 
                update = effect.split("=")[1].strip() 
                
                if ("(" in state_var):
                    state_var_args = state_var.split("(")[1].split(")")[0].split(",")
                    state_var_args = [arg.strip().lower() for arg in state_var_args]
                else:
                    state_var_args = []
                
                # Strip ? prefix from state_var_args for comparison, since
                # action_args already have ? stripped (e.g., "clue" vs "?clue")
                state_var_args_clean = [a.replace('?', '') for a in state_var_args]
                missing_args = []
                for arg in action_args:
                    if (arg.lower() not in state_var_args_clean):
                        missing_args.append(arg.lower())
                
                if (len(missing_args) > 0):
                    exist_str = "exists_{"
                    for arg in missing_args:
                        # Use canonical class type for RDDL quantifier
                        arg_type = arg_to_class.get(arg, arg)
                        exist_str += f"?{arg}:{arg_type}, "
                    exist_str = exist_str[:-2] + "} "
                else :
                    exist_str = None

                if probability is not None:
                    try:
                        prob_val = float(probability)
                        if prob_val < 1.0:
                            old_val_str = state_var
                            if "(" in old_val_str:
                                args_list = ["?" + arg for arg in state_var_args]
                                old_val_str = old_val_str.split("(")[0] + "(" + ", ".join(args_list) + ")"
                            
                            if update.lower() == "true":
                                update = f"(Bernoulli({prob_val}) | {old_val_str})"
                            elif update.lower() == "false":
                                stay_true_prob = round(1.0 - prob_val, 4)
                                update = f"(Bernoulli({stay_true_prob}) & {old_val_str})"
                    except ValueError:
                        pass

                # Look up CPF: try exact match first, then fall back to base-name match
                # (handles "is_collected_clue(?clue)" matching "is_collected_clue(Clue)")
                sv_base = state_var.split("(")[0] if "(" in state_var else state_var
                matched_key = state_var if state_var in sv_cpfs else sv_base_to_key.get(sv_base)
                if matched_key and matched_key in sv_cpfs:
                    cpfs = sv_cpfs[matched_key]
                    branch = Branch()
                    branch._set_exist_str(exist_str)
                    branch._add_cond(precond)
                    branch._set_rhs(update)
                    cpfs._add_branch(branch)
        
        for key,val in sv_cpfs.items():
            branch = Branch()
            if ("(" in key):
                args = key.split("(")[1].split(")")[0].split(",") 
                args = ["?"+arg.strip().lower() for arg in args]
                key = key.split("(")[0] + "(" + ",".join(args) + ")"
            branch._set_rhs(key)
            val._add_branch(branch)
            cpfss.append(val)   
            
        return cpfss
    
    def _make_rddl_var(self, original, raw_name):
        """Return the RDDL variable reference for a name.

        If the original YAML text uses a ? prefix (e.g., ?eco_engineer), it is
        always treated as a typed variable even if a domain object with the same
        name exists.  Only bare names (no ? prefix) that match a known domain
        object are treated as concrete instance references.
        """
        if original.startswith('?'):
            # Explicitly marked as a variable in the YAML — keep ? prefix
            return f"?{raw_name}"
        if raw_name in self.obj_to_class:
            # Bare domain object reference — no ? prefix
            return raw_name
        # Unknown bare name — treat as variable
        return f"?{raw_name}"

    def _resolve_class(self, name, arg_to_class):
        """Resolve a name to its canonical class type.

        Checks: arg_to_class mapping first, then domain-object-to-class mapping,
        then falls back to the name itself.
        """
        lower = name.lower()
        if lower in arg_to_class:
            return arg_to_class[lower]
        if lower in self.obj_to_class:
            return self.obj_to_class[lower]
        return lower

    def modify_prec(self, prec, arg_to_class=None):
        if arg_to_class is None:
            arg_to_class = {}
        prec = prec.replace("NOT ", "~")
        prec = prec.replace(" AND ", " & ")

        # Translate Object.x == Object.x syntax to RDDL 'exists' statements
        match_x = re.search(r'([A-Za-z0-9_\?]+)\.x\s*==\s*([A-Za-z0-9_\?]+)\.x', prec)
        if match_x:
            a, b = match_x.groups()
            a_raw, b_raw = a.replace('?', '').lower(), b.replace('?', '').lower()
            a_class = self._resolve_class(a_raw, arg_to_class)
            b_class = self._resolve_class(b_raw, arg_to_class)
            # Domain objects (concrete instances) should NOT get ? prefix
            a_var = self._make_rddl_var(a, a_raw)
            b_var = self._make_rddl_var(b, b_raw)
            prec = prec.replace(match_x.group(0), f"exists_{{?x:xpos}} [{a_class}-x({a_var}, ?x) & {b_class}-x({b_var}, ?x)]")

        match_y = re.search(r'([A-Za-z0-9_\?]+)\.y\s*==\s*([A-Za-z0-9_\?]+)\.y', prec)
        if match_y:
            a, b = match_y.groups()
            a_raw, b_raw = a.replace('?', '').lower(), b.replace('?', '').lower()
            a_class = self._resolve_class(a_raw, arg_to_class)
            b_class = self._resolve_class(b_raw, arg_to_class)
            a_var = self._make_rddl_var(a, a_raw)
            b_var = self._make_rddl_var(b, b_raw)
            prec = prec.replace(match_y.group(0), f"exists_{{?y:ypos}} [{a_class}-y({a_var}, ?y) & {b_class}-y({b_var}, ?y)]")

        return prec
         
    def _get_action_preconditions(self, clss):
        preconds = []
        precond = f"""forall_{{?{clss}:{clss}}}[
            move-left-{clss}(?{clss}) => exists_{{?x1:xpos, ?x2:xpos}}[{clss}-x(?{clss}, ?x1) & ADJACENT-LEFT(?x1, ?x2)] 
        ];"""
        preconds.append(precond)

        precond = f"""forall_{{?{clss}:{clss}}}[
            move-right-{clss}(?{clss}) => exists_{{?x1:xpos, ?x2:xpos}}[{clss}-x(?{clss}, ?x1) & ADJACENT-RIGHT(?x1, ?x2)] 
        ];"""
        preconds.append(precond)

        precond = f"""forall_{{?{clss}:{clss}}}[
            move-up-{clss}(?{clss}) => exists_{{?y1:ypos, ?y2:ypos}}[{clss}-y(?{clss}, ?y1) & ADJACENT-UP(?y1, ?y2)] 
        ];"""
        preconds.append(precond)

        precond = f"""forall_{{?{clss}:{clss}}}[
            move-down-{clss}(?{clss}) => exists_{{?y1:ypos, ?y2:ypos}}[{clss}-y(?{clss}, ?y1) & ADJACENT-DOWN(?y1, ?y2)] 
        ];"""
        preconds.append(precond)

        return preconds
        

    def get_action_preconditions(self):
        preconds = []
        for clss in self.parsed_data["classes"]:
            if (self.parsed_data["classes"][clss]["isMovable"]):
                preconds.extend(self._get_action_preconditions(clss))
        return preconds
    
    
    def _quantify_predicates(self, expr):
        """Find predicates with arguments in an expression and wrap them with
        forall quantifiers so they are valid RDDL.

        For example:
            is_at_landmark_player_landmark(player, ?landmark)
        becomes:
            forall_{?player:player, ?landmark:landmark} [is_at_landmark_player_landmark(?player, ?landmark)]
        """
        def _replace_pred(match):
            full = match.group(0)
            pred_name = match.group(1)
            args_str = match.group(2)

            # Skip if this looks like an arithmetic function or comparison
            if pred_name in ('Bernoulli', 'abs', 'min', 'max', 'floor', 'ceil',
                             'round', 'log', 'exp', 'pow', 'sqrt', 'if'):
                return full

            # Look up the state variable definition to find argument types
            sv_info = None
            for sv_key in self.parsed_data.get("state_variables", {}):
                if "(" in sv_key and sv_key.split("(")[0] == pred_name:
                    sv_info = sv_key
                    break

            if sv_info is None:
                return full  # Not a known state-variable predicate, leave as-is

            # Get the declared argument types from the state variable definition
            sv_type_args = sv_info.split("(")[1].split(")")[0].split(",")
            sv_type_args = [a.strip() for a in sv_type_args]

            args = [a.strip() for a in args_str.split(',')]

            quant_vars = []
            new_args = []
            for i, arg in enumerate(args):
                clean = arg.replace('?', '').strip()
                # Determine the RDDL type for this argument
                if i < len(sv_type_args):
                    arg_type = sv_type_args[i]
                else:
                    arg_type = clean
                var_name = f"?{clean}"
                quant_vars.append(f"{var_name}:{arg_type}")
                new_args.append(var_name)

            new_pred = f"{pred_name}({', '.join(new_args)})"
            if quant_vars:
                return f"forall_{{{', '.join(quant_vars)}}} [{new_pred}]"
            return new_pred

        # Match predicate calls: word(args) — but not things like == or arithmetic
        return re.sub(r'([a-zA-Z_][a-zA-Z0-9_]*)\(([^)]+)\)', _replace_pred, expr)

    def get_termination(self):
        goal = self.parsed_data["goal"]
        goal = goal.replace("NOT ", " ~")
        goal = goal.replace("AND", " & ")
        goal = goal.replace("OR", " | ")
        goal = self._quantify_predicates(goal)
        return goal

    def get_reward(self):
        reward = self.parsed_data["reward"]
        reward = reward.replace("NOT ", " ~")
        reward = reward.replace("AND", " & ")
        reward = reward.replace("OR", " | ")
        reward = self._quantify_predicates(reward)
        return reward

    def get_grid_size(self):
        x_size = self.parsed_data["grid_size"]["x_size"]
        y_size = self.parsed_data["grid_size"]["y_size"]
        return x_size, y_size

    def _get_obj_pos(self, clss, obj):
        for key,val in self.parsed_data["initial_state_objects"].items():
            if (key == obj):
                x = val["position"]["x"]
                y = val["position"]["y"]
                # Add 'x' and 'y' prefixes to match the RDDL objects (e.g., x1, y2)
                return f"{clss}-x({obj}, x{x})", f"{clss}-y({obj}, y{y})"

    def get_non_fluents(self):
        non_fluents = []
        for i in range(1,self.x_size):
            non_fluents.append(f"ADJACENT-LEFT(x{i+1}, x{i})")
        for i in range(1,self.x_size):
            non_fluents.append(f"ADJACENT-RIGHT(x{i}, x{i+1})")
        
        for i in range(1,self.y_size):
            non_fluents.append(f"ADJACENT-UP(y{i+1}, y{i})")
        for i in range(1,self.y_size):
            non_fluents.append(f"ADJACENT-DOWN(y{i}, y{i+1})")

        for clss in self.parsed_data["classes"]:
            if (not self.parsed_data["classes"][clss]["isMovable"]):
                objs = self.parsed_data["classes"][clss]["instances"]
                for obj in objs:
                    x,y = self._get_obj_pos(clss, obj)
                    non_fluents.append(x)
                    non_fluents.append(y)
        
        return non_fluents

    def get_init_state(self):
        init_state = [] 
        for clss in self.parsed_data["classes"]:
            if (self.parsed_data["classes"][clss]["isMovable"]):
                objs = self.parsed_data["classes"][clss]["instances"]
                for obj in objs:
                    x,y = self._get_obj_pos(clss, obj)
                    init_state.append(x)
                    init_state.append(y)
        
        for key,val in self.parsed_data["initial_state_variables"].items():
            if (type(val) == bool and val == True):
                init_state.append(f"{key}")
            elif (type(val) == bool and val == False):
                init_state.append(f"~{key}")
            else:
                init_state.append(f"{key} = {val}")
        
        return init_state


    def render_instance(self):
        self.instance = f"instance {self.name} {{ \n\n"
        self.instance += "\t domain = " + self.name + ";\n\n"
        self.instance += "\t objects {\n"
        
        x_pos = [f"x{i}" for i in range(1, self.x_size+1)]
        y_pos = [f"y{i}" for i in range(1, self.y_size+1)]
        self.instance += "\t\t xpos : { " + ", ".join(x_pos) + " };\n"
        self.instance += "\t\t ypos : { " + ", ".join(y_pos) + " };\n"
        for clss in self.parsed_data["classes"]:
            instances_str = "{ " + ", ".join(self.parsed_data["classes"][clss]["instances"]) + " };" 
            self.instance += f"\t\t {clss} : {instances_str}\n"
        self.instance += "\t };\n\n"

        self.instance += "\t non-fluents {\n"
        for nf in self.non_fluents:
            self.instance += f"\t\t {nf};\n"
        self.instance += "\t };\n\n"

        self.instance += "\t init-state {\n"
        for state in self.init_state:
            self.instance += f"\t\t {state};\n" 
        self.instance += "\t };\n\n"

        self.instance += "max-nondef-actions = 1;\n"
        self.instance += "horizon = 100;\n"
        self.instance += "discount = 0.80 ;\n"

        self.instance += "}\n"
        return self.instance


def write_file_with_directories(file_path, content):
    path = Path(file_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, 'w') as f:
        f.write(content)
